// Types & layouts (fp32 baseline)
using ElementInputA      = float;
using ElementInputB      = float;
using ElementOutput      = float;
using ElementAccumulator = float;

using LayoutA = cutlass::layout::RowMajor;     // A(MxK) row-major
using LayoutB = cutlass::layout::ColumnMajor;  // Reinterpret B(NxK row-major) as (KxN col-major) to represent B^T without copy
using LayoutC = cutlass::layout::RowMajor;     // C(MxN) row-major

using GemmOp = cutlass::gemm::device::Gemm<
    ElementInputA, LayoutA,
    ElementInputB, LayoutB,
    ElementOutput, LayoutC,
    ElementAccumulator
>;

// Computes C = A @ B^T
// A: [M, K] (row-major contiguous)
// B: [N, K] (row-major contiguous), but consumed as (K, N) column-major to avoid an explicit transpose
torch::Tensor ab_bt_gemm(torch::Tensor A, torch::Tensor B) {
  TORCH_CHECK(A.is_cuda() && B.is_cuda(), "A and B must be CUDA tensors");
  TORCH_CHECK(A.dim()==2 && B.dim()==2,  "A and B must be 2D");
  TORCH_CHECK(A.scalar_type()==at::kFloat && B.scalar_type()==at::kFloat,
              "This build expects float32 tensors");

  int64_t M64 = A.size(0);
  int64_t K64 = A.size(1);
  TORCH_CHECK(B.size(1) == K64, "B.shape[1] (K) must match A.shape[1] (K)");
  int64_t N64 = B.size(0);

  int M = static_cast<int>(M64);
  int K = static_cast<int>(K64);
  int N = static_cast<int>(N64);

  auto A_c = A.contiguous();
  auto B_c = B.contiguous();
  auto C   = torch::empty({M64, N64}, A.options());

  // Leading dimensions by layout
  int ldA = K; // RowMajor leading dimension = columns
  int ldB = K; // ColumnMajor leading dimension = rows (we view B as (K x N) column-major)
  int ldC = N; // RowMajor leading dimension = columns

  auto ptrA = reinterpret_cast<ElementInputA const*>(A_c.data_ptr<float>());
  auto ptrB = reinterpret_cast<ElementInputB const*>(B_c.data_ptr<float>());
  auto ptrC = reinterpret_cast<ElementOutput*      >(C.data_ptr<float>());

  GemmOp gemm;
  typename GemmOp::Arguments args(
      {M, N, K},            // Problem size (M, N, K)
      {ptrA, ldA},          // A tensor ref
      {ptrB, ldB},          // B tensor ref (interpreted as ColumnMajor KxN)
      {ptrC, ldC},          // C tensor ref
      {ptrC, ldC},          // D tensor ref (output)
      {ElementAccumulator(1.0f), ElementAccumulator(0.0f)} // D = 1*A*B + 0*C
  );

}
