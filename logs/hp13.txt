__device__ inline float bf16_to_float(bf16 v) {

}
__device__ inline bf16 float_to_bf16(float v) {

__device__ inline float silu(float x) {
  return x / (1.0f + __expf(-x));
}

// Kernel: fused SwiGLU MLP
// x:  [M, D]
// w1: [H, D]
// w3: [H, D]
// w2: [D, H]  (Linear with in_features=H, out_features=D; PyTorch stores (out,in))
// y:  [M, D]
__global__ void fused_swiglu_mlp_kernel_bf16(
    const bf16* __restrict__ x,
    const bf16* __restrict__ w1,
    const bf16* __restrict__ w3,
    const bf16* __restrict__ w2,
    bf16* __restrict__ y,
    int M, int D, int H)
{
  extern __shared__ unsigned char smem_raw[];
  bf16* sX = reinterpret_cast<bf16*>(smem_raw);
  bf16* sG = reinterpret_cast<bf16*>(smem_raw + size_t(D) * sizeof(bf16));

  int row = blockIdx.x;
  if (row >= M) return;

  // Pointers to current row in x and y
  const bf16* x_row = x + size_t(row) * D;
  bf16* y_row = y + size_t(row) * D;

  // Load x_row into shared memory
  for (int k = threadIdx.x; k < D; k += blockDim.x) {
    sX[k] = x_row[k];
  }
  __syncthreads();

  // 1) Compute gate g = silu(W1 * x) * (W3 * x), store in sG as bf16
  // Each thread processes a strided set of H rows
  for (int h = threadIdx.x; h < H; h += blockDim.x) {
    const bf16* __restrict__ w1_row = w1 + size_t(h) * D;
    const bf16* __restrict__ w3_row = w3 + size_t(h) * D;
    float acc1 = 0.f;
    float acc3 = 0.f;
    // Dot products over D
    for (int k = 0; k < D; ++k) {
      float xv  = bf16_to_float(sX[k]);
      float w1v = bf16_to_float(w1_row[k]);
      float w3v = bf16_to_float(w3_row[k]);
      acc1 += xv * w1v;
      acc3 += xv * w3v;
    }
    float g = silu(acc1) * acc3;
    sG[h] = float_to_bf16(g);
  }
  __syncthreads();

  // 2) Compute y_row = sG @ W2^T
  // W2 stored as [D, H] row-major; row d contiguous along H
  for (int d = threadIdx.x; d < D; d += blockDim.x) {
    const bf16* __restrict__ w2_row = w2 + size_t(d) * H;
    float acc = 0.f;
    for (int h = 0; h < H; ++h) {
      float gv = bf16_to_float(sG[h]);
      float wv = bf16_to_float(w2_row[h]);
      acc += gv * wv;
    }
    y_row[d] = float_to_bf16(acc);
  }
}

// x:  [M, D]
// w1: [H, D]
// w3: [H, D]

    const bf16* __restrict__ x,
    const bf16* __restrict__ w1,
    const bf16* __restrict__ w3,
    const bf16* __restrict__ w2,
    bf16* __restrict__ y,
    int M, int D, int H)
{
  extern __shared__ unsigned char smem_raw[];
  bf16* sX = reinterpret_cast<bf16*>(smem_raw);
  bf16* sG = reinterpret_cast<bf16*>(smem_raw + size_t(D) * sizeof(bf16));

  int row = blockIdx.x;
  if (row >= M) return;

  const bf16* x_row = x + size_t(row) * D;
  bf16* y_row = y + size_t(row) * D;

  // Each thread processes a strided set of H rows
  for (int h = threadIdx.x; h < H; h += blockDim.x) {

      float w1v = bf16_to_float(w1_row[k]);
      float w3v = bf16_to_float(w3_row[k]);

      acc3 += xv * w3v;

    sG[h] = float_to_bf16(g);
