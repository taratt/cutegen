torch::Tensor atb_gemm(torch::Tensor A, torch::Tensor B) {
    TORCH_CHECK(A.is_cuda() && B.is_cuda(),

        {M, N, K},          // Problem size
        {ptrA, lda},        // A
        {ptrB, ldb},        // B
        {ptrC, ldc},        // C (output)

        {ElementAccumulator(1.0f), ElementAccumulator(0.0f)}   // alpha, beta

using LayoutB = cutlass::layout::RowMajor;
using LayoutC = cutlass::layout::RowMajor;

    TORCH_CHECK(A.is_cuda() && B.is_cuda(),
                "atb_gemm expects CUDA tensors");

                "A and B must be 2-D tensors");
    TORCH_CHECK(A.scalar_type() == at::kFloat && B.scalar_type() == at::kFloat,
                "Currently only supports float32 tensors");

    int M = static_cast<int>(A.size(1));

    auto B_c = B.contiguous();

    int ldb = N;  // RowMajor  leading dimension (cols)

    auto ptrB = reinterpret_cast<ElementInputB const*>(B_c.data_ptr<float>());
